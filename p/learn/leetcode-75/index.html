public class Solution {
    public String mergeAlternately(String word1, String word2) {
        int maxLen = Math.max(word1.length(), word2.length());
        StringBuilder str = new StringBuilder();
        for (int i=0; i<maxLen; i++) {
            if (i < word1.length()) {
                str.append(word1.charAt(i));
            }
            if (i < word2.length()) {
                str.append(word2.charAt(i));
            }
        }
        return str.toString();
    }
}

public class Solution {
    public String gcdOfStrings(String str1, String str2) {
        int gcdLen = gcdLen(str1.length(), str2.length());
        String gcdStr = str1.substring(0, gcdLen);
        if (checkGcd(str1, gcdStr) && checkGcd(str2, gcdStr)) {
            return gcdStr;
        }
        return "";
    }

    public int gcdLen(int a, int b) {
        int remainder = a % b;
        while (remainder != 0) {
            a = b;
            b = remainder;
            remainder = a % b;
        }
        return b;
    }

    public boolean checkGcd(String str, String gcdStr) {
        int multiple = str.length() / gcdStr.length();
        StringBuilder result = new StringBuilder();
        for (int i=0 ; i<multiple; i++) {
            result.append(gcdStr);
        }
        return str.equals(result.toString());
    }
}
// 时间复杂度：O(n)

public class Solution {

        int maxCandie = candies[0];
        for (int i=1; i<candies.length; i++) {
            maxCandie = Math.max(maxCandie, candies[i]);

1.4 种花问题

【种花问题】

  • 假设有一个很长的花坛，一部分地块种植了花，另一部分却没有。可是，花不能种植在相邻的地块上，它们会争夺水源，两者都会死去。
  • 给你一个整数数组flowerbed表示花坛，由若干0和1组成，其中0表示没种植花，1表示种植了花。另有一个数n，能否在不打破种植规则的情况下种入n朵花？能则返回true，不能则返回false。
  • 解题思路
    • 总结每种空地能多少花的规律
    • （1）空地左右两边有花，例10001，能种花的数量为(空地长度-1)/2
    • （2）左or右边界没有花，例00001，能种花的数量为空地长度/2
    • （3）空地完全没有花，例00000，能种花的数量为(空地长度+1)/2
    • 总结以上规律后，利用指针计算出空地的长度，再而计算能种花的数量，最后对比n

public class Solution {
    public boolean canPlaceFlowers(int[] flowerbed, int n) {
        int prev = -1;
        int placeCount = 0;
        for (int i=0; i<flowerbed.length; i++) {
            if (flowerbed[i] == 1) {
                if (prev < 0) {
                    // 空地在左边界
                    placeCount += i / 2;
                } else {
                    // 空地在中间
                    placeCount += (i - prev - 2) / 2;
                }
                prev = i;
            }
        }
        if (prev < 0) {
            // 全是空地
            placeCount = (flowerbed.length + 1) / 2;
        } else {
            // 空地在右边界
            placeCount += (flowerbed.length - prev - 1) / 2;
        }
        return placeCount >= n;
    }
}

public class Solution {
  public String reverseVowels(String s) {
    char[] charArray = s.toCharArray();
    int left = 0;
    int right = s.length() - 1;
    while (left < right) {
      while (left < s.length() && !isVowel(charArray[left])) {
        left ++;
      }
      while (right > -1 && !isVowel(charArray[right])) {
        right --;
      }
      if (left < right) {
        char temp = charArray[left];
        charArray[left] = charArray[right];
        charArray[right] = temp;
        left ++;
        right --;
      }
    }
    return new String(charArray);
  }

  public boolean isVowel(char c) {
    return "aoiueAOIUE".indexOf(c) >= 0;
  }
}
// 时间复杂度：O(n)

public class Solution {
  public String reverseWords(String s) {
    int start = -1;
    StringBuilder result = new StringBuilder();
    for (int i=0; i<s.length(); i++) {
      // 空格，切割单词并反向拼接
      if (start != -1 && s.charAt(i) == ' ') {
        result.insert(0, s.substring(start, i) + " ");
        start = -1;
        continue;
      }
      // 非空格，记录单词起始位置
      if (start == -1 && s.charAt(i) != ' ') {
        start = i;
      }
    }
    // 边界处理
    if (start != -1) {
      result.insert(0, s.substring(start) + " ");
    }
    return result.substring(0, result.length()-1);
  }
}

1.7 除自身以外数组的乘积

【除自身以外数组的乘积】

  • 给你一个整数数组nums，返回数组answer，其中answer[i]等于nums中除nums[i]之外其余各元素的乘积。
  • 题目数据保证数组nums之中任意元素的全部前缀元素和后缀的乘积都在32位整数范围内。
  • 请不要使用除法，且在O(n)时间复杂度内完成此题。
  • 解题思路
    • 因为题目禁止除法，以及禁止嵌套循环，需要思路非寻常解法
    • 准备从左到右累乘的数组leftMul，以及从右到左累乘的数组rigthMul，隔一位进行累乘
    • 最后结果就是leftMul[i] * rightMul[i]，相当与把num[i]的值改为1的累乘

public class Solution {
    public int[] productExceptSelf(int[] nums) {
        int n = nums.length;
        int[] leftMul = new int[n];
        // 计算左乘数组
        leftMul[0] = 1;
        for (int i=1; i<n; i++) {
            leftMul[i] = nums[i-1] * leftMul[i-1];
        }
        // 计算右乘数组
        int[] rightMul = new int[n];
        rightMul[n-1] = 1;
        for (int i=n-2; i>-1; i--) {
            rightMul[i] = nums[i+1] * rightMul[i+1];
        }
        // 计算结果
        int[] result = new int[n];
        for (int i=0; i<n; i++) {
            result[i] = leftMul[i] * rightMul[i];
        }
        return result;
    }
}

1.8 递增的三元子序列

【递增的三元子序列】

  • 给你一个整数数组nums，判断这个数组中是否存在长度为3的递增子序列。
  • 如果存在这样的三元组下标(i, j, k)且满足i < j < k，使得nums[i] < nums[j] < nums[k]，返回true；否则，返回false。
  • 解题思路
    • 维护两个数组，从左到右的最小值数组leftMin，和从右到左的最大值数组rightMax
    • 遍历数组，保证leftMin[i-1] < nums[i] < rightMax[i+1]

public class Solution {
    public boolean increasingTriplet(int[] nums) {
        int n = nums.length;
        int[] leftMin = new int[n];
        leftMin[0] = nums[0];
        int[] rightMax = new int[n];
        rightMax[n-1] = nums[n-1];
        for (int i=1; i<n; i++) {
            leftMin[i] = Math.min(leftMin[i-1], nums[i]);
        }
        for (int i=n-2; i>-1; i--) {
            rightMax[i] = Math.max(rightMax[i+1], nums[i]);
        }
        for (int i=1; i<n-1; i++) {
            if (leftMin[i-1] < nums[i] && rightMax[i+1] > nums[i]) {
                return true;
            }
        }
        return false;
    }
}

1.9 压缩字符串

public class Solution {
    public int compress(char[] chars) {
        int strLen = 0;
        int start = 0;
        for (int i=0; i<chars.length; i++) {
            if (i == chars.length -1 || chars[i] != chars[i+1]) {
                chars[strLen++] = chars[i];
                int size = i - start + 1;
                // 长度大于1，后面补充长度字符
                if (size > 1) {
                    char[] nums = String.valueOf(size).toCharArray();
                    for (char num : nums) {
                        chars[strLen++] = num;
                    }
                }
                start = i + 1;
            }
        }
        return strLen;
    }
}

public class Solution {
    public void moveZeroes(int[] nums) {
        int cur = 0;
        // 非0元素左移
        for (int i=0; i<nums.length; i++) {
            if (nums[i] != 0) {
                nums[cur++] = nums[i];
            }
        }
        // 剩下元素改写为0
        for (int i=cur; i<nums.length; i++) {
            nums[i] = 0;
        }
    }
}

public class Solution {
    public boolean isSubsequence(String s, String t) {
        if (s.length() > t.length()) {
            return false;
        }
        if (s.isEmpty()) {
            return true;
        }
        int charIdx = 0;
        for (int i=0; i<t.length(); i++) {
            if (t.charAt(i) != s.charAt(charIdx)) {
                continue;
            }
            charIdx ++;
            if (charIdx == s.length()) {
                return true;
            }
        }
        return false;
    }
}

public class Solution {
    public int maxArea(int[] height) {
        int left = 0;
        int right = height.length-1;
        int size = 0;
        while (left < right) {
            int minHeight = Math.min(height[left], height[right]);
            size = Math.max(size, (right-left) * minHeight);
            if (height[left] < height[right]) {
                while (height[left] <= minHeight) {
                    left ++;
                }
            } else {
                while (height[right] <= minHeight) {
                    right --;
                }
            }
        }
        return size;
    }
}

        int step = 0;
        while (left < right) {
            if (nums[left] + nums[right] == k) {
                left ++;
                right --;
                step ++;
            } else if (nums[left] + nums[right] > k) {
                right --;
            } else {
                left ++;
            }
        }
        return step;
    }
}
// 时间复杂度：O(n)

class Solution {
    public boolean closeStrings(String word1, String word2) {
        if (word1.length() != word2.length()) {
            return false;
        }
        int[] count1 = new int[26];
        int[] count2 = new int[26];

            count1[c - 'a']++;

            count2[c - 'a']++;

            return Collections.emptyList();

    public void getRightList(Queue<TreeNode> nodes, List<Integer> rightList) {

        for (int i=0; i<n; i++) {
            TreeNode node = nodes.poll();
            if (!Objects.isNull(node.left)) {
                nodes.offer(node.left);
            }
            if (!Objects.isNull(node.right)) {
                nodes.offer(node.right);
            }
            if (i == n-1) {
                rightList.add(node.val);
            }
        }
        this.getRightList(nodes, rightList);
    }
}

        for (int i = 1; i <= chars1.length; i++) {
            for (int j = 1; j <= chars2.length; j++) {
                if (chars1[i] == chars2[j]) {
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                } else {
                    dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
                }
            }
        }
        return dp[text1.length() - 1][text2.length() - 1];
    }
}
// 时间复杂度：O(mn)

10.5 买卖股票的最佳时机含手续费

买卖股票的最佳时机含手续费

  • 给定一个整数数组 prices，其中 prices[i] 表示第 i 天的股票价格 ；整数 fee 代表了交易股票的手续费用。
  • 你可以无限次地完成交易，但是你每笔交易都需要付手续费。如果你已经购买了一个股票，在卖出它之前你就不能再继续购买股票了。
  • 返回获得利润的最大值。
  • 注意：这里的一笔交易指买入持有并卖出股票的整个过程，每笔交易你只需要为支付一次手续费。
  • 示例：
  • 输入： prices = [1, 3, 2, 8, 4, 9], fee = 2
  • 输出： 8
  • 解释： 能够达到的最大利润:
    • 在此处买入 prices[0] = 1
    • 在此处卖出 prices[3] = 8
    • 在此处买入 prices[4] = 4
    • 在此处卖出 prices[5] = 9
    • 总利润: ((8 - 1) - 2) + ((9 - 4) - 2) = 8
  • 解题思路：维护一个二维dp数组，或者两个一维dp数组，其中一个记录无股票时的金额，另一个记录有股票的金额
    • dp[0][0] = 0; dp[1][0] = -prices[i];
    • dp[0][i] = max(dp[0][i - 1], prices[i] + dp[1][i - 1] - fee)，即手里无股票，有可能是前一天也没股票；或者前一天有股票，但在今天卖出
    • dp[1][i] = max(dp[0][i - 1] - prices[i], dp[1][i - 1])，即手里有股票，有可能是前一天没股票，在今天进行购入；或者前一天就有股票了

class Solution {
    public int maxProfit(int[] prices, int fee) {
        int[][] dp = new int[2][prices.length];
        dp[0][0] = 0;
        dp[1][0] = -prices[0];
        for (int i = 1; i < prices.length; i++) {
            dp[0][i] = Math.max(dp[0][i - 1], prices[i] + dp[1][i - 1] - fee);
            dp[1][i] = Math.max(dp[0][i - 1] - prices[i], dp[1][i - 1]);
        }
        return dp[0][prices.length - 1];
    }
}
// 时间复杂度：O(n)

10.6 编辑距离

编辑距离

  • 给你两个单词 word1 和 word2， 请返回将 word1 转换成 word2 所使用的最少操作数 。
  • 你可以对一个单词进行如下三种操作：
    • 插入一个字符
    • 删除一个字符
    • 替换一个字符
  • 示例 1：
  • 输入： word1 = "horse", word2 = "ros"
  • 输出： 3
  • 解释：
    • horse -> rorse (将 'h' 替换为 'r')
    • rorse -> rose (删除 'r')
    • rose -> ros (删除 'e')
  • 解题思路： 维护一个二维的dp数组，里面记录字符a到字符b的最小编辑距离
    • dp[0][i] = i, dp[j][0] = j，初始化，空字符到字符a，或字符b的最小编辑距离是字符长度
    • dp[i][j] = min(dp[i - 1][j] + 1, dp[i][j - 1] + 1, dp[i - 1][j - 1])，字符的最短距离，是前一个字符的最短 + 1；如何字符i，字符j相同时，最短距离则为前一个字符，无需+1
    • dp[i][j] = min(dp[i - 1][j] + 1, dp[i][j - 1] + 1, dp[i - 1][j - 1] + 1)，同理，只是字符i和字符j不相同时，需要+1编辑距离

class Solution {
    public int minDistance(String word1, String word2) {
        int m = word1.length();
        int n = word2.length();
        int[][] dp = new int[m + 1][n + 1];
        for (int i = 0; i < m + 1; i++) {
            dp[i][0] = i;
        }
        for (int i = 0; i < n + 1; i++) {
            dp[0][i] = i;
        }
        char[] char1 = word1.toCharArray();
        char[] char2 = word2.toCharArray();
        for (int i = 1; i < m + 1; i++) {
            for (int j = 1; j < n + 1; j++) {
                if (char1[i - 1] == char2[j - 1]) {
                    dp[i][j] = Math.min(Math.min(dp[i - 1][j] + 1, dp[i][j - 1] + 1), dp[i - 1][j - 1]);
                } else {
                    dp[i][j] = Math.min(Math.min(dp[i - 1][j] + 1, dp[i][j - 1] + 1), dp[i - 1][j - 1] + 1);
                }
            }
        }
        return dp[m][n];
    }
}
// 时间复杂度：O(mn)
